%% ── Save a copy of the .toc for the back-matter detailed ToC ──────
%% \tableofcontents (via \@starttoc) clears the .toc after reading it,
%% so we duplicate it to a .dtoc file before that happens.  The
%% back-matter detailed ToC reads .dtoc with tocdepth=3.
%%
%% tex.jobname returns '"Foundational knowledge"' (with quotes) when
%% the filename contains spaces.  Strip the embedded quotes so that
%% Lua's io.open receives a bare filename.
\directlua{
  local jn = tex.jobname:gsub('"', '')
  local src = io.open(jn .. ".toc", "r")
  if src then
    local dst = io.open(jn .. ".dtoc", "w")
    if dst then
      dst:write(src:read("*a"))
      dst:close()
    end
    src:close()
  end
}
